cmake_minimum_required(VERSION 3.8)
project(rover_usb_camera)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-app-1.0)
foreach(target IN ITEMS usb_camera webcam_test)
  add_executable(${target} src/${target}.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_link_libraries(${target} PkgConfig::GST)
  ament_target_dependencies(${target} rclcpp sensor_msgs)
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
